    #[parallel(cuda)]
    from a == 0 { a += 1; } until a == 3;
    #[concurrent]
    { a += 1; }
    try { a += 1; } catch_rollback { a -= 1; } -> failed;
    logged h { a = 1; }
    irrev { a = 2; }
